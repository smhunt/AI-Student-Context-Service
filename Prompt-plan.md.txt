Product: StudentContext AI

Author: Sean Hunt
Date: February 2026

1. Overview
StudentContext is a middleware SaaS that transforms any school-board-approved LLM chatbot into a personalized learning companion by maintaining a vectorized knowledge base of each student’s complete digital academic history. It provides role-based contextual augmentation for students (self-service learning) and staff (classroom intelligence, report writing, guidance).
Target customer: Ontario school boards, starting with Thames Valley DSB.
Pricing: $1/student/month (10 months), $2/staff/month (12 months).
Tech stack: TypeScript, Node.js, PostgreSQL + pgvector, React, Claude API (swappable).

4.2 Consent Verification
Every context retrieval also checks consent:

// src/services/consent.ts

async function verifyConsent(
  studentId: string,
  requestedSources: DocumentSource[]
): Promise<{ allowed: boolean; filteredSources: DocumentSource[] }> {
  
  const consent = await db.consent_records.findActive(studentId, 'ai_context');
  
  if (!consent || consent.status !== 'granted') {
    return { allowed: false, filteredSources: [] };
  }

  // Filter to only consented data sources
  const filteredSources = requestedSources.filter(
    source => consent.data_sources.includes(source)
  );

  return { allowed: true, filteredSources };
}


5. Context Engine
The heart of the product — takes a user’s chat query and returns an augmented prompt with relevant student context.

// src/services/context-engine.ts

interface ContextRequest {
  userId: string;              // who is asking
  query: string;               // what they asked
  targetStudentId?: string;    // staff mode: which student
  courseId?: string;            // optional course scope
  maxChunks?: number;          // default 10
}

interface AugmentedPrompt {
  systemPrompt: string;
  userMessage: string;
  chunksUsed: string[];        // chunk IDs for audit
}

async function buildAugmentedPrompt(req: ContextRequest): Promise<AugmentedPrompt> {
  
  // 1. Resolve permissions
  const user = await getUser(req.userId);
  const scope = PERMISSION_RULES[user.role](user);
  
  // 2. Determine target student(s)
  const targetStudentId = req.targetStudentId || 
    (user.role === 'student' ? user.id : null);
  
  if (targetStudentId && !scope.studentIds.includes(targetStudentId)) {
    throw new ForbiddenError('No access to this student');
  }

  // 3. Check consent
  const consent = await verifyConsent(targetStudentId, scope.dataSources);
  if (!consent.allowed) {
    // Return generic (non-augmented) prompt
    return buildGenericPrompt(req.query);
  }

  // 4. Embed the query
  const queryEmbedding = await embedText(req.query);

  // 5. Vector similarity search scoped to student + permissions
  const relevantChunks = await db.query(`
    SELECT c.id, c.content, c.metadata, 
           1 - (e.embedding <=> $1) as similarity
    FROM embeddings e
    JOIN chunks c ON c.id = e.chunk_id
    JOIN documents d ON d.id = c.document_id
    WHERE e.student_id = $2
      AND c.sensitivity <= $3
      AND d.source = ANY($4)
    ORDER BY e.embedding <=> $1
    LIMIT $5
  `, [
    queryEmbedding,
    targetStudentId,
    scope.sensitivityMax,
    consent.filteredSources,
    req.maxChunks || 10,
  ]);

  // 6. Build the augmented system prompt
  const contextBlock = relevantChunks.map(chunk => 
    `[${chunk.metadata.source} | ${chunk.metadata.date}]\n${chunk.content}`
  ).join('\n\n');

  const systemPrompt = buildSystemPrompt(user.role, targetStudentId, contextBlock);

  // 7. Audit log
  await logAudit({
    actorId: req.userId,
    action: 'context_retrieval',
    targetStudentId,
    details: {
      query: req.query,
      chunksRetrieved: relevantChunks.map(c => c.id),
      chunkCount: relevantChunks.length,
    },
  });

  return {
    systemPrompt,
    userMessage: req.query,
    chunksUsed: relevantChunks.map(c => c.id),
  };
}

function buildSystemPrompt(
  role: UserRole, 
  studentId: string, 
  context: string
): string {
  
  if (role === 'student') {
    return `You are a personalized learning assistant for a student in an Ontario school.
You have access to this student's academic history and context. Use it to provide
personalized, encouraging, and educationally appropriate responses. Meet the student
where they are based on their demonstrated knowledge and skills.

Never reveal raw grades, scores, or sensitive information directly. Instead, use this
context to calibrate your explanations and examples to their level.

STUDENT CONTEXT:
${context}

GUIDELINES:
- Reference specific topics they've studied when relevant
- Build on concepts they've demonstrated understanding of
- Identify and gently address gaps in prerequisite knowledge
- Use encouraging, age-appropriate language
- If asked about something outside their curriculum, connect it to what they know

  }

  if (role === 'teacher') {

Provide data-informed insights while maintaining professional boundaries.

${context}

- Maintain professional, clinical language

  }

  // ... similar for guidance_counsellor, supply_teacher, etc.
}

// src/ingestion/google-classroom.ts

interface GoogleClassroomSync {
  boardId: string;
  schoolId: string;
  accessToken: string;  // OAuth token from board's Google admin
}

async function syncGoogleClassroom(config: GoogleClassroomSync) {
  const classroom = google.classroom({ version: 'v1', auth: config.accessToken });

  // 1. Sync courses
  const courses = await classroom.courses.list({ courseStates: ['ACTIVE'] });
  
  for (const course of courses.data.courses) {
    const dbCourse = await upsertCourse({
      schoolId: config.schoolId,
      externalId: course.id,
      name: course.name,
      source: 'google_classroom',
      metadata: { section: course.section, room: course.room },
    });

    // 2. Sync roster
    const students = await classroom.courses.students.list({ courseId: course.id });
    const teachers = await classroom.courses.teachers.list({ courseId: course.id });
    
    await syncRoster(dbCourse.id, students.data.students, teachers.data.teachers);

    // 3. Sync coursework and submissions
    const coursework = await classroom.courses.courseWork.list({ courseId: course.id });
    
    for (const work of coursework.data.courseWork) {
      const submissions = await classroom.courses.courseWork.studentSubmissions.list({
        courseId: course.id,
        courseWorkId: work.id,
      });

      for (const submission of submissions.data.studentSubmissions) {
        await ingestSubmission(dbCourse, work, submission);
      }
    }
  }
}

async function ingestSubmission(course, assignment, submission) {
  const studentId = await resolveStudentId(submission.userId);
  
  // Build document content from submission
  const content = buildSubmissionContent(assignment, submission);
  
  // Dedup check
  const hash = sha256(content);
  const existing = await db.documents.findByHash(hash);
  if (existing) return;

  // Create document
  const doc = await db.documents.create({
    studentId,
    boardId: course.boardId,
    source: 'google_classroom_submission',
    sourceId: submission.id,
    title: assignment.title,
    content,
    contentDate: submission.updateTime,
    courseId: course.id,
    sensitivity: 'standard',
    metadata: {
      assignmentType: assignment.workType,
      grade: submission.assignedGrade,
      maxGrade: assignment.maxPoints,
      state: submission.state,
      late: submission.late,
    },
    hash,
  });

  // Chunk and embed
  await chunkAndEmbed(doc);
}

// src/ingestion/embedder.ts

const CHUNK_SIZE = 500;      // tokens
const CHUNK_OVERLAP = 50;    // token overlap between chunks

async function chunkAndEmbed(document: Document) {
  // 1. Split content into chunks
  const chunks = splitIntoChunks(document.content, CHUNK_SIZE, CHUNK_OVERLAP);

  for (let i = 0; i < chunks.length; i++) {
    // 2. Store chunk
    const chunk = await db.chunks.create({
      documentId: document.id,
      studentId: document.studentId,
      boardId: document.boardId,
      content: chunks[i].text,
      chunkIndex: i,
      tokenCount: chunks[i].tokenCount,
      sensitivity: document.sensitivity,
      metadata: {
        ...document.metadata,
        source: document.source,
        date: document.contentDate,
        title: document.title,
      },
    });

    // 3. Generate embedding
    const embedding = await generateEmbedding(chunks[i].text);

    // 4. Store embedding
    await db.embeddings.create({
      chunkId: chunk.id,
      studentId: document.studentId,
      boardId: document.boardId,
      embedding,
      model: 'text-embedding-3-small',
      sensitivity: document.sensitivity,
    });
  }
}

async function generateEmbedding(text: string): Promise<number[]> {
  // Use OpenAI embeddings (good quality, cheap, fast)
  // The CHAT model can be Claude while embeddings use OpenAI — this is common
  const response = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: text,
  });
  return response.data[0].embedding;
}

11. Key Technical Decisions
Embedding model: OpenAI text-embedding-3-small (1536 dimensions). Cheap ($0.02/1M tokens), fast, good quality. The chat LLM can be anything — embeddings are separate.
Vector index: pgvector HNSW. At 82K students × ~100 documents each × ~5 chunks each = ~41M embeddings at full board scale. HNSW handles this on a single PostgreSQL instance with sufficient RAM (64GB at board scale). Pilot scale (2K students) is trivial.
LLM abstraction: The llm-adapter.ts service wraps multiple providers behind a common interface. Start with Claude, but the board might mandate Copilot or Gemini. The adapter pattern keeps the context engine provider-agnostic.
Multi-tenancy: Board-scoped at the data level (every table has board_id). This lets you run multiple boards on one infrastructure while maintaining strict data isolation. Row-level security in PostgreSQL can enforce this at the database level as a defense-in-depth measure.
Sync strategy: Initial full sync + incremental via webhooks (Google Classroom supports push notifications). SIS sync via nightly batch until real-time integration is available. All syncs are idempotent (hash-based dedup).
